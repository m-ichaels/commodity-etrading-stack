cmake_minimum_required(VERSION 3.20)
project(commodity_etrading_stack CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(MSVC)
  add_compile_options(/W3 /EHsc /permissive- /Zc:__cplusplus /utf-8)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
  add_compile_options(-Wall -Wextra)
endif()
add_library(cetk INTERFACE)
target_include_directories(cetk INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(cetk_cli src/main.cpp)
target_link_libraries(cetk_cli PRIVATE cetk)
set_target_properties(cetk_cli PROPERTIES OUTPUT_NAME cetk)
add_executable(cetk_fixgw src/fix_gateway.cpp)
target_link_libraries(cetk_fixgw PRIVATE cetk)
add_executable(cetk_tests tests/tests.cpp)
target_link_libraries(cetk_tests PRIVATE cetk)
if(WIN32)
  target_link_libraries(cetk_fixgw PRIVATE ws2_32)
  target_link_libraries(cetk_tests PRIVATE ws2_32)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(cetk_fixgw PRIVATE Threads::Threads)
  target_link_libraries(cetk_tests PRIVATE Threads::Threads)
endif()
enable_testing()
add_test(NAME cetk_tests COMMAND cetk_tests)
